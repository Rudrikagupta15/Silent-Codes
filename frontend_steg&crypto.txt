    <script>
        document.addEventListener("DOMContentLoaded", () => {
            // --- Cryptography Section ---
            const cryptoTextarea = document.getElementById('crypto-text');
            const cryptoKeyInput = document.getElementById('crypto-key');
            const cryptoResultTextarea = document.getElementById('crypto-result');
            const encryptBtn = document.getElementById('encrypt-btn');
            const decryptBtn = document.getElementById('decrypt-btn');

            // Simple Vigenère cipher implementation
            const encryptDecrypt = (text, key, isEncrypt) => {
                if (!key) return "[ERROR] AUTH_KEY required for CRYPTO operation.";
                let result = '';
                const base = 'A'.charCodeAt(0);
                const a = 'a'.charCodeAt(0);
                const keyLen = key.length;

                for (let i = 0; i < text.length; i++) {
                    let char = text[i];
                    if (/[a-zA-Z]/.test(char)) {
                        let isUpperCase = char === char.toUpperCase();
                        let charCode = isUpperCase ? char.charCodeAt(0) - base : char.charCodeAt(0) - a;
                        let keyChar = key[i % keyLen];
                        let keyCharCode = keyChar.toUpperCase().charCodeAt(0) - base;
                        let newCharCode;

                        if (isEncrypt) {
                            newCharCode = (charCode + keyCharCode) % 26;
                        } else {
                            newCharCode = (charCode - keyCharCode + 26) % 26;
                        }

                        char = isUpperCase ? String.fromCharCode(newCharCode + base) : String.fromCharCode(newCharCode + a);
                    }
                    result += char;
                }
                return result;
            };

            encryptBtn.addEventListener('click', () => {
                const text = cryptoTextarea.value;
                const key = cryptoKeyInput.value.trim();
                cryptoResultTextarea.value = encryptDecrypt(text, key, true);
            });

            decryptBtn.addEventListener('click', () => {
                const text = cryptoTextarea.value;
                const key = cryptoKeyInput.value.trim();
                cryptoResultTextarea.value = encryptDecrypt(text, key, false);
            });

            // --- Steganography Section ---
            const stegoFileInput = document.getElementById('stego-file');
            const stegoMessageInput = document.getElementById('stego-message');
            const stegoStatusText = document.getElementById('stego-status');
            const hideBtn = document.getElementById('hide-btn');
            const extractBtn = document.getElementById('extract-btn');
            const canvas = document.getElementById('stego-canvas');
            const downloadLink = document.getElementById('download-link');
            const ctx = canvas.getContext('2d');

            let originalImage = new Image();

            // Handle file selection and image rendering
            stegoFileInput.addEventListener('change', (e) => {
                const file = e.target.files[0];
                if (!file || !file.type.startsWith('image/')) {
                    stegoStatusText.textContent = "[ERROR] Please select a valid image file.";
                    return;
                }
                const reader = new FileReader();
                reader.onload = (event) => {
                    originalImage.onload = () => {
                        canvas.width = originalImage.width;
                        canvas.height = originalImage.height;
                        ctx.drawImage(originalImage, 0, 0);
                        stegoStatusText.textContent = `[SUCCESS] Carrier file loaded: ${file.name}`;
                        downloadLink.style.display = 'none';
                        // Show the canvas once an image is loaded
                        canvas.classList.remove('hidden');
                    };
                    originalImage.src = event.target.result;
                };
                reader.readAsDataURL(file);
            });

            // LSB (Least Significant Bit) encoding and decoding logic
            const LSB = {
                encode: (imageData, message) => {
                    const pixels = imageData.data;
                    const binaryMessage = LSB.toBinary(message + "###"); // Add a delimiter
                    if (binaryMessage.length > pixels.length) {
                        return "[ERROR] Message too long for this image.";
                    }

                    for (let i = 0; i < binaryMessage.length; i++) {
                        const bit = binaryMessage[i];
                        pixels[i] = (pixels[i] & 0xFE) | parseInt(bit);
                    }
                    return pixels;
                },
                decode: (imageData) => {
                    const pixels = imageData.data;
                    let binaryMessage = "";
                    for (let i = 0; i < pixels.length; i++) {
                        binaryMessage += (pixels[i] & 1).toString();
                        if (binaryMessage.endsWith(LSB.toBinary("###"))) {
                            return LSB.fromBinary(binaryMessage.slice(0, -LSB.toBinary("###").length));
                        }
                    }
                    return "No hidden message found.";
                },
                toBinary: (str) => {
                    return str.split('').map(char => {
                        return ('0000000' + char.charCodeAt(0).toString(2)).slice(-8);
                    }).join('');
                },
                fromBinary: (str) => {
                    return str.match(/.{1,8}/g).map(bin => {
                        return String.fromCharCode(parseInt(bin, 2));
                    }).join('');
                }
            };

            // Event listener to hide data in the image
            hideBtn.addEventListener('click', () => {
                if (!originalImage.src) {
                    stegoStatusText.textContent = "[ERROR] No carrier image selected.";
                    return;
                }
                const message = stegoMessageInput.value;
                if (!message) {
                    stegoStatusText.textContent = "[ERROR] No hidden message provided.";
                    return;
                }

                ctx.drawImage(originalImage, 0, 0); // Redraw clean image
                const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                const encodedData = LSB.encode(imageData, message);

                if (typeof encodedData === 'string') {
                    stegoStatusText.textContent = encodedData;
                    return;
                }

                ctx.putImageData(imageData, 0, 0);
                
                downloadLink.style.display = 'inline';
                downloadLink.href = canvas.toDataURL("image/png");
                downloadLink.download = "encoded_image.png";
                stegoStatusText.textContent = "[SUCCESS] Data hidden. Download link activated.";
            });

            // Event listener to extract data from the image
            extractBtn.addEventListener('click', () => {
                if (!originalImage.src) {
                    stegoStatusText.textContent = "[ERROR] No carrier image selected.";
                    return;
                }

                const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                const extractedMessage = LSB.decode(imageData);
                stegoMessageInput.value = extractedMessage;
                stegoStatusText.textContent = `[SUCCESS] Extracted message: "${extractedMessage}"`;
            });
        });
    </script>
